2. **Group KPIs by Category**:
```tsx
const groupedKpis = useMemo(() => {
  const filtered = kpis.filter(kpi => 
    selectedValueChain === 'all' || 
    KPI_CATEGORIES.find(c => c.id === kpi.categoryId)?.valueChain === selectedValueChain
  );
  
  return KPI_CATEGORIES
    .filter(cat => selectedValueChain === 'all' || cat.valueChain === selectedValueChain)
    .map(category => ({
      ...category,
      kpis: filtered.filter(kpi => kpi.categoryId === category.id)
    }))
    .filter(group => group.kpis.length > 0);
}, [kpis, selectedValueChain]);
```
